/* ===================== HELPERS ===================== */
function normalizeHeader(h){
  return h.toString().normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase().replace(/[^a-z0-9]/g,'');
}
function toNumber(v){
  if(v===null || v===undefined || v==='') return 0;
  if(typeof v === 'number') return v;
  let s = String(v).trim();
  if(s.includes(',')) s = s.replace(/\./g,'').replace(',', '.');
  const n = parseFloat(s);
  return isNaN(n) ? 0 : n;
}
function round2(n){ return Math.round(n*100)/100; }
function parseDate(v){
  if(!v) return null;
  if(v instanceof Date && !isNaN(v)) return v;
  const s = String(v).trim();
  let m = s.match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{2,4})$/);
  if(m){ let [,d,mo,y]=m; if(y.length===2) y='20'+y; return new Date(+y,+mo-1,+d); }
  m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/);
  if(m){ const [,y,mo,d]=m; return new Date(+y,+mo-1,+d); }
  const d2 = new Date(s);
  return isNaN(d2) ? null : d2;
}
function defaultPontos(key){
  switch(key){
    case 'agua_diaria': return 1;
    case 'dias_treinados': return 5;
    case 'indicacoes': return 20;
    default: return 1;
  }
}

/* ===================== LEITURA DO ARQUIVO ===================== */
function mapRow(row){
  const mapped = {};
  for(const rawKey in row){
    const nk = normalizeHeader(rawKey);
    if(nk.includes('idvenda') || (nk.includes('id') && nk.includes('venda'))) mapped.id_venda = row[rawKey];
    else if(nk.includes('datapagamento') || (nk.includes('data') && nk.includes('pagamento'))) mapped.data_pagamento = row[rawKey];
    else if(nk.includes('agua')) mapped.agua_diaria = row[rawKey];
    else if(nk.includes('diastreinados') || (nk.includes('dias') && nk.includes('trein'))) mapped.dias_treinados = row[rawKey];
    else if(nk.includes('indica')) mapped.indicacoes = row[rawKey];
    else if(nk.includes('treino')) mapped.treino = row[rawKey];
  }
  return mapped;
}

function handleFile(file){
  if(!file) return;
  const isCSV = /\.csv$/i.test(file.name);
  const reader = new FileReader();
  reader.onload = function(evt){
    try{
      const data = evt.target.result;
      const wb = XLSX.read(data, { type: isCSV ? 'string' : 'array', cellDates:true });
      const sheet = wb.Sheets[wb.SheetNames[0]];
      const json = XLSX.utils.sheet_to_json(sheet, {defval:''});
      if(!json.length){ alert('A planilha está vazia.'); return; }
      processUpload(json, file.name);
    }catch(err){
      console.error(err);
      alert('Erro ao ler o arquivo. Verifique se o formato é válido.');
    }
  };
  if(isCSV) reader.readAsText(file,'UTF-8'); else reader.readAsArrayBuffer(file);
}

function processUpload(json, fileName){
  const mapped = json.map(mapRow).filter(r=>r.id_venda !== undefined && r.id_venda !== '');
  if(!mapped.length){
    alert('Não consegui identificar a coluna "id_venda" no arquivo. Verifique os nomes das colunas.');
    return;
  }
  state.rawData = mapped.map((r,i)=>({...r, _rowId:i, _dataObj: parseDate(r.data_pagamento)}));
  document.getElementById('file-status-text').textContent = `📄 ${fileName} — ${state.rawData.length} registros carregados.`;
  document.getElementById('btn-change-file').classList.remove('hidden');
  document.getElementById('last-update').textContent = 'Atualizado em ' + new Date().toLocaleString('pt-BR');

  detectAndBuildConfig();
  populateFilterOptions();
  calculateScores();
  state.page = 1;
  applyFilters();
  showSections();
}

/* ===================== DETECÇÃO DE TIPO + CONFIG ===================== */
function detectType(key){
  const vals = state.rawData.map(r=>r[key]).filter(v=>v!==undefined && v!==null && v!=='');
  if(!vals.length) return 'numeric';
  let numericCount = 0;
  vals.forEach(v=>{
    if(typeof v === 'number') numericCount++;
    else if(/^[\d.,\s]+$/.test(String(v).trim()) && !isNaN(toNumber(v))) numericCount++;
  });
  return (numericCount / vals.length) >= 0.9 ? 'numeric' : 'categorical';
}

function loadSavedConfig(){
  try{ return JSON.parse(localStorage.getItem(CONFIG_STORAGE_KEY)) || {}; }
  catch(e){ return {}; }
}
function saveConfig(){ localStorage.setItem(CONFIG_STORAGE_KEY, JSON.stringify(state.config)); }

function detectAndBuildConfig(){
  const saved = loadSavedConfig();
  const newConfig = {};
  COLUMN_DEFS.forEach(def=>{
    const type = detectType(def.key);
    const prev = saved[def.key];
    if(type === 'numeric'){
      newConfig[def.key] = {
        label: def.label, type,
        enabled: prev?.enabled ?? true,
        weight: prev?.weight ?? 1,
        pontosPorUnidade: prev?.pontosPorUnidade ?? defaultPontos(def.key)
      };
    } else {
      const uniqueVals = [...new Set(state.rawData.map(r=>String(r[def.key] ?? '').trim()).filter(v=>v!==''))];
      const valores = {};
      uniqueVals.forEach(v=>{ valores[v] = prev?.valores?.[v] ?? 0; });
      newConfig[def.key] = {
        label: def.label, type,
        enabled: prev?.enabled ?? true,
        weight: prev?.weight ?? 1,
        valores
      };
    }
  });
  state.config = newConfig;
  saveConfig();
  renderConfigPanel();
}

function resetConfig(){
  if(!confirm('Restaurar a pontuação padrão? Suas edições serão perdidas.')) return;
  localStorage.removeItem(CONFIG_STORAGE_KEY);
  detectAndBuildConfig();
  calculateScores();
  applyFilters();
}

function renderConfigPanel(){
  const container = document.getElementById('config-body');
  container.innerHTML = COLUMN_DEFS.map(def=>{
    const cfg = state.config[def.key];
    if(!cfg) return '';
    let bodyHTML = `<label class="cfg-label">Peso (importância)
        <input type="number" class="cfg-input cfg-weight" data-key="${def.key}" min="0" max="10" step="0.1" value="${cfg.weight}">
      </label>`;
    if(cfg.type === 'numeric'){
      bodyHTML += `<label class="cfg-label">Pontos por unidade
          <input type="number" class="cfg-input cfg-ppu" data-key="${def.key}" step="0.1" value="${cfg.pontosPorUnidade}">
        </label>`;
    } else {
      const keys = Object.keys(cfg.valores);
      const rows = keys.map(v=>`
        <div class="cfg-value-row">
          <span title="${v}">${v}</span>
          <input type="number" class="cfg-input cfg-value" data-key="${def.key}" data-value="${v}" step="1" value="${cfg.valores[v]}">
        </div>`).join('');
      bodyHTML += `<label class="cfg-label">Pontos por valor</label><div class="cfg-values-list">${rows || '<p style="color:var(--text-dim);font-size:12px">Nenhum valor encontrado.</p>'}</div>`;
    }
    return `<div class="config-card">
      <div class="config-card-header">
        <label class="switch">
          <input type="checkbox" class="cfg-input cfg-enabled" data-key="${def.key}" ${cfg.enabled ? 'checked' : ''}>
          <span class="slider"></span>
        </label>
        <div>
          <h4>${cfg.label}</h4>
          <span class="badge">${cfg.type === 'numeric' ? 'Numérico' : 'Categórico'}</span>
        </div>
      </div>
      <div class="config-card-body">${bodyHTML}</div>
    </div>`;
  }).join('');
}

/* ===================== CÁLCULO DE PONTUAÇÃO ===================== */
function calculateScores(){
  state.processedData = state.rawData.map(row=>{
    let score = 0;
    const breakdown = {};
    COLUMN_DEFS.forEach(def=>{
      const cfg = state.config[def.key];
      if(!cfg || !cfg.enabled){ breakdown[def.key] = 0; return; }
      let pts = 0;
      if(cfg.type === 'numeric'){
        pts = toNumber(row[def.key]) * cfg.pontosPorUnidade * cfg.weight;
      } else {
        const v = String(row[def.key] ?? '').trim();
        const base = cfg.valores[v] ?? 0;
        pts = base * cfg.weight;
      }
      breakdown[def.key] = round2(pts);
      score += pts;
    });
    return {...row, _score: round2(score), _breakdown: breakdown};
  });
  state.processedData.sort((a,b)=>b._score - a._score);
  state.processedData.forEach((r,i)=>r._rank = i+1);
}

function populateFilterOptions(){
  const sel = document.getElementById('filter-treino');
  const unique = [...new Set(state.rawData.map(r=>String(r.treino ?? '').trim()).filter(v=>v!==''))].sort();
  sel.innerHTML = '<option value="">Todos os treinos</option>' + unique.map(v=>`<option value="${v}">${v}</option>`).join('');

function applyFilters(){
  const search = document.getElementById('search-input').value.trim().toLowerCase();
  const treinoF = document.getElementById('filter-treino').value;
  const dateStart = document.getElementById('filter-date-start').value;
  const dateEnd = document.getElementById('filter-date-end').value;

  state.filteredData = state.processedData.filter(r=>{
    if(search && !String(r.id_venda).toLowerCase().includes(search)) return false;
    if(treinoF && String(r.treino).trim() !== treinoF) return false;
    if(dateStart){ const ds = new Date(dateStart); if(!r._dataObj || r._dataObj < ds) return false; }
    if(dateEnd){ const de = new Date(dateEnd); de.setHours(23,59,59,999); if(!r._dataObj || r._dataObj > de) return false; }
    return true;
  });

  sortData(false);
  populateStudentSelect();
  renderKPIs();
  renderCharts();
  renderTable();
}

/* ===================== ORDENAÇÃO DA TABELA ===================== */
function sortData(toggle=true){
  const {field} = state.sort;
  if(toggle) state.sort.dir = state.sort.dir === 'asc' ? 'desc' : 'asc';
  const dir = state.sort.dir === 'asc' ? 1 : -1;
  state.filteredData.sort((a,b)=>{
    let va = a[field], vb = b[field];
    if(field === '_dataObj'){ va = a._dataObj ? a._dataObj.getTime() : 0; vb = b._dataObj ? b._dataObj.getTime() : 0; }
    else if(typeof va === 'string' || typeof vb === 'string'){ va = String(va ?? '').toLowerCase(); vb = String(vb ?? '').toLowerCase(); }
    if(va < vb) return -1 * dir;
    if(va > vb) return 1 * dir;
    return 0;
  });
}

/* ===================== KPIs ===================== */
function renderKPIs(){
  const data = state.filteredData;
  const total = data.length;
  const media = total ? round2(data.reduce((a,r)=>a+r._score,0)/total) : 0;
  const melhor = [...data].sort((a,b)=>b._score-a._score)[0];
  const totalIndic = data.reduce((a,r)=>a+toNumber(r.indicacoes),0);
  const mediaDias = total ? round2(data.reduce((a,r)=>a+toNumber(r.dias_treinados),0)/total) : 0;

  document.getElementById('kpi-total').textContent = total;
  document.getElementById('kpi-media').textContent = media.toFixed(1);
  document.getElementById('kpi-melhor').textContent = melhor ? `ID ${melhor.id_venda} (${melhor._score.toFixed(1)} pts)` : '-';
  document.getElementById('kpi-indicacoes').textContent = totalIndic;
  document.getElementById('kpi-dias').textContent = mediaDias.toFixed(1);
}

function renderDoughnut(){
  const cfg = state.config.treino;
  let labels=[], data=[];
  if(cfg && cfg.type === 'categorical'){
    const counts = {};
    state.filteredData.forEach(r=>{
      const v = String(r.treino ?? '').trim() || 'Não informado';
      counts[v] = (counts[v]||0)+1;
    });
    labels = Object.keys(counts); data = Object.values(counts);
  } else {
    const vals = state.filteredData.map(r=>toNumber(r.treino));
    const max = Math.max(0, ...vals) || 1;
    labels = ['Baixo','Médio','Alto'];
    data = [
      vals.filter(v=>v < max*0.33).length,
      vals.filter(v=>v >= max*0.33 && v < max*0.66).length,
      vals.filter(v=>v >= max*0.66).length
    ];
  }
  const palette = ['#10b981','#34d399','#6ee7b7','#059669','#047857','#f2c94c','#a7f3d0','#0d9488'];
  if(state.charts.doughnut) state.charts.doughnut.destroy();
  state.charts.doughnut = new Chart(document.getElementById('doughnutChart').getContext('2d'), {
    type:'doughnut',
    data:{ labels, datasets:[{ data, backgroundColor: labels.map((_,i)=>palette[i%palette.length]), borderColor:'#0a1f19', borderWidth:2 }] },
    options:{ responsive:true, maintainAspectRatio:false, plugins:{ legend:{ position:'bottom', labels:{color:'#cfe9df', boxWidth:12} } } }
  });
}

function renderLineChart(){
  const map = {};
  state.filteredData.forEach(r=>{
    const d = r._dataObj;
    const key = d ? `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}` : null;
    if(!key) return;
    if(!map[key]) map[key] = {sum:0, count:0};
    map[key].sum += r._score; map[key].count++;
  });
  const keys = Object.keys(map).sort();
  const meses = ['Jan','Fev','Mar','Abr','Mai','Jun','Jul','Ago','Set','Out','Nov','Dez'];
  const labels = keys.map(k=>{ const [y,m]=k.split('-'); return meses[+m-1]+'/'+y.slice(2); });
  const dataAvg = keys.map(k=>round2(map[k].sum/map[k].count));
  const dataCount = keys.map(k=>map[k].count);

  if(state.charts.line) state.charts.line.destroy();
  state.charts.line = new Chart(document.getElementById('lineChart').getContext('2d'), {
    type:'bar',
    data:{
      labels,
      datasets:[
        { type:'line', label:'Pontuação média', data:dataAvg, borderColor:'#34d399', backgroundColor:'rgba(52,211,153,.15)', tension:.35, fill:true, yAxisID:'y' },
        { type:'bar', label:'Nº de registros', data:dataCount, backgroundColor:'rgba(242,201,76,.35)', yAxisID:'y1', borderRadius:6 }
      ]
    },
    options:{
      responsive:true, maintainAspectRatio:false,
      plugins:{ legend:{ labels:{color:'#cfe9df'} } },
      scales:{
        x:{ ticks:{color:'#cfe9df'}, grid:{display:false} },
        y:{ position:'left', ticks:{color:'#cfe9df'}, grid:{color:'rgba(255,255,255,.06)'} },
        y1:{ position:'right', ticks:{color:'#cfe9df'}, grid:{display:false} }
      }
    }
  });
}

function computeAverageBreakdown(){
  const cats = COLUMN_DEFS.filter(d=>state.config[d.key]?.enabled);
  const avg = {};
  cats.forEach(d=>{
    const sum = state.filteredData.reduce((acc,r)=>acc+(r._breakdown[d.key]||0),0);
    avg[d.key] = state.filteredData.length ? round2(sum/state.filteredData.length) : 0;
  });
  return avg;
}

function updateRadarChart(){
  const select = document.getElementById('student-select');
  const id = select.value;
  const student = state.filteredData.find(r=>String(r.id_venda) === String(id)) || state.filteredData[0];
  const cats = COLUMN_DEFS.filter(d=>state.config[d.key]?.enabled);
  const labels = cats.map(d=>state.config[d.key].label);
  const studentData = student ? cats.map(d=>student._breakdown[d.key] || 0) : cats.map(()=>0);
  const avgData = Object.values(computeAverageBreakdown());

  if(state.charts.radar) state.charts.radar.destroy();
  state.charts.radar = new Chart(document.getElementById('radarChart').getContext('2d'), {
    type:'radar',
    data:{
      labels,
      datasets:[
        { label: student ? 'Aluno ID '+student.id_venda : 'Aluno', data:studentData, backgroundColor:'rgba(16,185,129,.35)', borderColor:'#34d399', pointBackgroundColor:'#34d399' },
        { label:'Média do grupo', data:avgData, backgroundColor:'rgba(242,201,76,.15)', borderColor:'#f2c94c', pointBackgroundColor:'#f2c94c' }
      ]
    },
    options:{
      responsive:true, maintainAspectRatio:false,
      scales:{ r:{ angleLines:{color:'rgba(255,255,255,.08)'}, grid:{color:'rgba(255,255,255,.08)'}, pointLabels:{color:'#cfe9df'}, ticks:{display:false, backdropColor:'transparent'} } },

}

/* ===================== TABELA ===================== */
function renderTable(){
  const totalPages = Math.max(1, Math.ceil(state.filteredData.length / PAGE_SIZE));
  if(state.page > totalPages) state.page = totalPages;
  const start = (state.page - 1) * PAGE_SIZE;
  const pageItems = state.filteredData.slice(start, start + PAGE_SIZE);
  const tbody = document.getElementById('table-body');

  if(!pageItems.length){
    tbody.innerHTML = `<tr><td colspan="8" style="text-align:center;padding:30px;color:var(--text-dim)">Nenhum registro encontrado com os filtros atuais.</td></tr>`;
  } else {
    tbody.innerHTML = pageItems.map(r=>{
      let medal = '';
      if(r._rank === 1) medal = '🥇'; else if(r._rank === 2) medal = '🥈'; else if(r._rank === 3) medal = '🥉';
      const dataFmt = r._dataObj ? r._dataObj.toLocaleDateString('pt-BR') : '-';
      return `<tr>
        <td>${medal} #${r._rank}</td>
        <td>${r.id_venda}</td>
        <td>${dataFmt}</td>
        <td>${r.treino ?? '-'}</td>
        <td>${r.agua_diaria ?? '-'}</td>
        <td>${r.dias_treinados ?? '-'}</td>
        <td>${r.indicacoes ?? '-'}</td>
        <td class="score-cell">${r._score.toFixed(1)}</td>
      </tr>`;
    }).join('');
  }
  document.getElementById('page-info').textContent = `Página ${state.page} de ${totalPages}`;
  document.getElementById('prev-page').disabled = state.page <= 1;
  document.getElementById('next-page').disabled = state.page >= totalPages;
}

/* ===================== EXPORTAR CSV ===================== */
function exportCSV(){
  if(!state.filteredData.length){ alert('Não há dados para exportar.'); return; }
  const headers = ['id_venda','data_pagamento','treino','agua_diaria','dias_treinados','indicacoes','pontuacao_total'];
  const rows = state.filteredData.map(r=>[
    r.id_venda, r._dataObj ? r._dataObj.toLocaleDateString('pt-BR') : '', r.treino, r.agua_diaria, r.dias_treinados, r.indicacoes, r._score
  ]);
  let csv = headers.join(';') + '\n' + rows.map(row=>row.map(v=>`"${String(v ?? '').replace(/"/g,'""')}"`).join(';')).join('\n');
  const blob = new Blob(['\uFEFF'+csv], {type:'text/csv;charset=utf-8;'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = 'ranking_alunos_emerald_night.csv';
  document.body.appendChild(a); a.click(); document.body.removeChild(a);
  URL.revokeObjectURL(url);
}
